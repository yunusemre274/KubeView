import datetime, hashlib, json, os, pathlib, shutil, signal, subprocess, time

root = pathlib.Path.cwd()
private = root / '.git/release-audit/sbom-real-build-20261115'
source = private / 'source'
out = root / 'artifacts/release-audit/ra-20261115/sbom-real-package-root'
commit = subprocess.check_output(['git', 'rev-parse', 'HEAD'], text=True).strip()
archive = private / 'committed-runtime-source.tar'
paths = subprocess.check_output(['git', 'ls-tree', '--name-only', commit, '--', 'backend', 'frontend', 'scripts', 'manual-lab', 'docs', 'deploy', 'LICENSE', 'README.md', 'MANUAL_TEST_GUIDE.md', '.gitignore'], text=True).splitlines()
assert {'backend', 'frontend', 'scripts', 'manual-lab', 'docs', 'deploy', 'README.md', 'MANUAL_TEST_GUIDE.md', '.gitignore'}.issubset(paths)
with archive.open('xb') as stream:
    subprocess.run(['git', 'archive', '--format=tar', commit, '--', *paths], stdout=stream, check=True)
subprocess.run(['/usr/bin/tar', '-xf', str(archive), '-C', str(source)], check=True)
env = {**os.environ, 'PATH': '/Users/yunusemre/.nvm/versions/node/v24.16.0/bin:/Users/yunusemre/go/pkg/mod/golang.org/toolchain@v0.0.1-go1.26.11.darwin-arm64/bin:/usr/bin:/bin:/opt/homebrew/bin',
       'GOTOOLCHAIN': 'local', 'GOPROXY': 'off', 'GOMAXPROCS': '2', 'GOMEMLIMIT': '2GiB',
       'UV_THREADPOOL_SIZE': '2', 'NODE_OPTIONS': '--max-old-space-size=2048',
       'GIT_DIR': str(root / '.git'), 'GIT_WORK_TREE': str(source), 'GIT_OPTIONAL_LOCKS': '0'}
argv = ['/Users/yunusemre/.nvm/versions/node/v24.16.0/bin/node', 'scripts/build-release.mjs',
        '--prepared', '--require-clean', f'--output-dir={private / "packages"}', 'darwin-arm64', 'linux-amd64']
result = {'stage': 'actual npm lock SBOM + Go BuildInfo + archive packaging', 'sourceCommit': commit,
          'sourceSnapshotSHA256': hashlib.sha256(archive.read_bytes()).hexdigest(),
          'argv': argv, 'startedAt': datetime.datetime.now(datetime.timezone.utc).isoformat(), 'status': 'FAIL',
          'scope': 'isolated source and dependency clone, no cluster/credential/provider; native runtime pending', 'automaticRetries': 0}
start = time.monotonic()
with (out / 'package.stdout').open('xb') as stdout, (out / 'package.stderr').open('xb') as stderr:
    proc = subprocess.Popen(argv, cwd=source, env=env, stdout=stdout, stderr=stderr, start_new_session=True)
    result['ownedPID'] = proc.pid
    peak = 0
    try:
        while proc.poll() is None:
            rows = subprocess.check_output(['/bin/ps', '-axo', 'pgid=,rss='], text=True).splitlines()
            rss = sum(int(fields[1]) for row in rows if len(fields := row.split()) == 2 and fields[0] == str(proc.pid)) * 1024
            peak = max(peak, rss)
            if time.monotonic() - start > 240 or rss > 4 * (1 << 30) or shutil.disk_usage(private).free < 10 * (1 << 30):
                raise RuntimeError('bounded time/RSS/disk safeguard')
            time.sleep(.25)
        result['exitCode'] = proc.returncode
        if proc.returncode != 0:
            raise RuntimeError('actual package command nonzero; first failure retained')
        build = json.loads((private / 'packages/build.json').read_text())
        assert build['commit'] == commit and build['dirty'] is False
        for target in ['darwin-arm64', 'linux-amd64']:
            folder = private / f'packages/kubeview_1.1.0_{target}'
            for name, digest in build['packages'][target]['sboms'].items():
                assert hashlib.sha256((folder / name).read_bytes()).hexdigest() == digest
                shutil.copyfile(folder / name, out / f'{target}-{name}')
            shutil.copyfile(folder / 'build-provenance.json', out / f'{target}-build-provenance.json')
        shutil.copyfile(private / 'packages/build.json', out / 'package-build.json')
        shutil.copyfile(private / 'packages/SHA256SUMS', out / 'package-SHA256SUMS')
        result.update(status='PASS', packages=build['packages'], frontendAssets=build['frontendAssets'], runtimeInputsDirty=build['dirty'],
                      note='Partial private checkout may report actual Go vcs.modified=true; runtime input hashes are independently bound. Not final release/platform acceptance.')
    except Exception as error:
        result['failure'] = str(error)[:250]
        raise
    finally:
        if proc.poll() is None and os.getpgid(proc.pid) == proc.pid:
            os.killpg(proc.pid, signal.SIGTERM)
            try: proc.wait(timeout=3)
            except subprocess.TimeoutExpired:
                os.killpg(proc.pid, signal.SIGKILL)
                proc.wait(timeout=3)
        result.update(finishedAt=datetime.datetime.now(datetime.timezone.utc).isoformat(), seconds=time.monotonic()-start,
                      sampledOwnedGroupPeakRSSBytes=peak, childExitCode=proc.returncode)
        (out / 'package-result.json').write_text(json.dumps(result, indent=2)+'\n')
