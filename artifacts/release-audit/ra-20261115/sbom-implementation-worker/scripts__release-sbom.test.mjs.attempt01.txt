const copy = value => structuredClone(value);
function fixture(t) {
  const root = mkdtempSync(path.join(os.tmpdir(), 'kubeview-sbom-fixture-'));
  t.after(() => rmSync(root, { recursive: true, force: true }));
  const put = (name, data) => { const file = path.join(root, name); mkdirSync(path.dirname(file), { recursive: true }); writeFileSync(file, data); return file; };
  put('frontend/package.json', JSON.stringify({ name: 'frontend-test', version: '1.1.0' }));
  put('frontend/package-lock.json', JSON.stringify({ lockfileVersion: 3, packages: { '': { name: 'frontend-test', version: '1.1.0' }, 'node_modules/foo': { version: '1.0.0' }, 'node_modules/foo/node_modules/foo': { version: '2.0.0' } } }));
  put('frontend/index.html', '<html>source</html>'); put('frontend/vite.config.js', '// config'); put('frontend/build/package-inventory.mjs', '// plugin');
  put('frontend/public/logo.svg', '<svg/>'); put('frontend/dist/logo.svg', '<svg/>');
  const app = put('frontend/src/app.js', 'local app');
  const one = put('frontend/node_modules/foo/index.js', 'first library');
  put('frontend/node_modules/foo/package.json', JSON.stringify({ name: 'foo', version: '1.0.0' }));
  const two = put('frontend/node_modules/foo/node_modules/foo/index.js', 'second library');
  put('frontend/node_modules/foo/node_modules/foo/package.json', JSON.stringify({ name: 'foo', version: '2.0.0' }));
  put('frontend/dist/assets/app.js', 'const bundled = 1;\n');
  const bundle = { 'assets/app.js': { type: 'chunk', code: 'const bundled = 1;\n', modules: { [app]: { renderedLength: 3 }, [one]: { renderedLength: 5 }, [two]: { renderedLength: 4 }, '\0commonjsHelpers.js': { renderedLength: 10 }, '/not-read/treeshaken.js': { renderedLength: 0 } } } };
  const inventory = collectPackageInventory(path.join(root, 'frontend'), bundle);
  put('frontend/dist/package-inventory.json', JSON.stringify(inventory));
  const assets = { 'logo.svg': sha256('<svg/>'), 'assets/app.js': sha256(bundle['assets/app.js'].code), 'package-inventory.json': sha256(JSON.stringify(inventory)) };
  const validated = validateFrontendInventory({ root, frontendAssets: assets });
  const binding = { version: '1.1.0', commit: 'a'.repeat(40), target: 'darwin-arm64', binarySHA256: 'b'.repeat(64), inventory: validated, frontendAssets: assets, packageJSONSHA256: validated.packageJSONSHA256, lockSHA256: validated.lockSHA256 };
  return { root, put, bundle, inventory, assets, binding };
}
const goInfo = { GoVersion: 'go1.26.11', Main: { Path: 'example.invalid/kubeview', Version: '(devel)' }, Deps: [{ Path: 'example.invalid/module', Version: 'v1.0.0', Sum: 'h1:YXVkaXQ=' }], Settings: [{ Key: 'GOOS', Value: 'darwin' }, { Key: 'GOARCH', Value: 'arm64' }, { Key: 'CGO_ENABLED', Value: '0' }, { Key: 'vcs.revision', Value: 'a'.repeat(40) }, { Key: 'vcs.modified', Value: 'false' }] };
const npmInfo = { bomFormat: 'CycloneDX', specVersion: '1.5', metadata: { component: { 'bom-ref': 'frontend@1.1.0' }, tools: [{ vendor: 'npm', name: 'cli', version: '11.13.0' }] }, components: [
  { type: 'library', name: 'foo', version: '1.0.0', 'bom-ref': 'foo@1.0.0', externalReferences: [{ url: 'https://credential.invalid/synthetic-private-url' }] },
  { type: 'library', name: 'build-tool', version: '2.0.0', 'bom-ref': 'build-tool@2.0.0', properties: [{ name: 'cdx:npm:package:development', value: 'true' }] },
], dependencies: [{ ref: 'frontend@1.1.0', dependsOn: ['foo@1.0.0', 'build-tool@2.0.0'] }, { ref: 'foo@1.0.0', dependsOn: [] }, { ref: 'build-tool@2.0.0', dependsOn: [] }] };

test('emitted module inventory separates real package versions, local/virtual and treeshaken modules', t => {
  const f = fixture(t);
  assert.deepEqual(f.inventory.packages.map(p => `${p.name}@${p.version}`), ['foo@1.0.0', 'foo@2.0.0']);
  assert.deepEqual(f.inventory.totals, { renderedModules: 4, localModules: 1, virtualModules: 1, unresolvedModules: 0 });
  assert.equal(JSON.stringify(f.inventory).includes(f.root), false);
  assert.equal(f.inventory.unresolved.length, 0);
  let emitted;
  packageInventory({ root: path.join(f.root, 'frontend') }).generateBundle.call({ emitFile: value => { emitted = value; } }, {}, f.bundle);
  assert.equal(emitted.fileName, 'package-inventory.json');
  assert.equal(JSON.parse(emitted.source).chunks['assets/app.js'].sha256, f.assets['assets/app.js']);
  assert.equal(f.bundle['assets/app.js'].code, 'const bundled = 1;\n');
});

test('unknown owner is counted and cannot satisfy exact bundled release coverage', t => {
  const f = fixture(t);
  f.bundle['assets/app.js'].modules['/outside/private/path.js'] = { renderedLength: 1 };
  const value = collectPackageInventory(path.join(f.root, 'frontend'), f.bundle);
  assert.equal(value.totals.unresolvedModules, 1); assert.equal(value.unresolved[0].reason, 'outside-frontend-root');
  assert.equal(JSON.stringify(value).includes('/outside/private'), false);
  f.put('frontend/dist/package-inventory.json', JSON.stringify(value));
  assert.throws(() => validateFrontendInventory({ root: f.root, frontendAssets: f.assets }), /unresolved/);
});

test('binary BOM binds actual module/stdLib/target and keeps source dev graph separate', t => {
  const f = fixture(t), runtime = binaryBom(goInfo, f.binding), source = frontendSourceBom(npmInfo, f.binding);
  assert.equal(runtime.metadata.component.hashes[0].content, f.binding.binarySHA256);
  assert.equal(runtime.components.find(c => c.name === 'stdlib').version, '1.26.11');
  assert.equal(runtime.components.filter(c => c.name === 'foo').length, 2);
  assert.equal(runtime.components.some(c => c.name === 'build-tool'), false);
  assert.ok(source.components.some(c => c.name === 'build-tool' && c.properties.some(p => p.name === 'kubeview:development' && p.value === 'true')));
  assert.equal(source.dependencies[0].dependsOn.length, 2);
  const serialized = JSON.stringify(source);
  assert.equal(serialized.includes('synthetic-private-url'), false);
  assert.equal(serialized.includes('(devel)'), false);
  assert.equal(runtime.components[0].properties.find(p => p.name === 'kubeview:goModuleH1').value, 'h1:YXVkaXQ=');
  assert.equal(runtime.components[0].hashes, undefined);
});

test('versioned Go replacements use actual identity; local replacements and target/revision mismatch fail', t => {
  const f = fixture(t), actual = copy(goInfo);
  actual.Deps[0].Replace = { Path: 'example.invalid/replacement', Version: 'v1.2.3', Sum: 'h1:YXVkaXQ=' };
  const bom = binaryBom(actual, f.binding);
  assert.equal(bom.components[0].name, 'example.invalid/replacement'); assert.equal(bom.components[0].version, 'v1.2.3');
  actual.Deps[0].Replace = { Path: '/private/local/module', Version: '' };
  assert.throws(() => binaryBom(actual, f.binding));
  assert.throws(() => binaryBom(goInfo, { ...f.binding, target: 'linux-amd64' }), /target/);
  assert.throws(() => binaryBom(goInfo, { ...f.binding, commit: 'c'.repeat(40) }), /revision/);
  const dirty = copy(goInfo); dirty.Settings.find(p => p.Key === 'vcs.modified').Value = 'true';
  assert.equal(binaryBom(dirty, f.binding).metadata.component.properties.find(p => p.name === 'kubeview:goVcsModified').value, 'true');
});

test('bounded malformed/missing/duplicate/dangling metadata is rejected, not empty clean fallback', t => {
  const f = fixture(t);
  for (const raw of ['{', '{}', ' '.repeat(MAX_JSON_BYTES + 1)]) assert.throws(() => binaryBom(raw, f.binding));
  const duplicate = copy(npmInfo); duplicate.components.push(copy(duplicate.components[0])); assert.throws(() => frontendSourceBom(duplicate, f.binding), /duplicate/);
  const edge = copy(npmInfo); edge.dependencies[0].dependsOn.push('not-a-component'); assert.throws(() => frontendSourceBom(edge, f.binding), /dangling/);
  const bom = binaryBom(goInfo, f.binding); bom.components.push(copy(bom.components[0])); assert.throws(() => validateBom(bom), /duplicate/);
  const privateRef = copy(npmInfo); privateRef.components[0]['bom-ref'] = 'https://user:credential@example.invalid/private'; assert.throws(() => frontendSourceBom(privateRef, f.binding), /non-public/);
});

test('npm collection uses explicit offline/no-script config, bounded executor, cleans temporary config on failure', t => {
  const f = fixture(t);
  let captured;
  const result = collectFrontendSource({ root: f.root, binding: f.binding, execute(command, args, options) {
    captured = { command, args, options };
    assert.equal(command, 'npm'); assert.equal(args[0], 'sbom'); assert.ok(args.includes('--package-lock-only')); assert.ok(args.includes('--include=dev')); assert.ok(args.includes('--offline')); assert.ok(args.includes('--ignore-scripts'));
    assert.equal(options.timeout, 15000); assert.equal(options.maxBuffer, MAX_JSON_BYTES);
    assert.equal(options.env.NODE_ENV, 'development'); assert.equal(options.env.NPM_CONFIG_OMIT, undefined);
    for (const flag of ['--userconfig', '--globalconfig']) assert.equal(readFileSync(args[args.indexOf(flag) + 1], 'utf8'), '');
    return JSON.stringify(npmInfo);
  } });
  assert.equal(result.components.length, 2);
  const directory = path.dirname(captured.args[captured.args.indexOf('--userconfig') + 1]); assert.equal(existsSync(directory), false);
  let errorDirectory;
  assert.throws(() => collectFrontendSource({ root: f.root, binding: f.binding, execute(_command, args) { errorDirectory = path.dirname(args[args.indexOf('--userconfig') + 1]); throw Error('synthetic timeout/private error omitted'); } }), /no archive published/);
  assert.equal(existsSync(errorDirectory), false);
});


test('prepared source/build/public and profiling metadata are current while unconsumed tests do not stale it', t => {
  const f = fixture(t);
  f.put('frontend/src/unconsumed.test.js', 'not a consumed Rollup module');
  assert.doesNotThrow(() => validateFrontendInventory({ root: f.root, frontendAssets: f.assets }));
  f.put('frontend/src/app.js', 'changed consumed module');
  assert.throws(() => validateFrontendInventory({ root: f.root, frontendAssets: f.assets }), /stale consumed/);
});
