const equalDeps = (a, b) =>
  a && b && a.length === b.length && a.every((x, i) => Object.is(x, b[i]));
const nodes = (node) => {
  if (Array.isArray(node)) return node.flatMap(nodes);
  if (!node || typeof node !== "object") return [];
  return [node, ...nodes(node.props?.children)];
};

function mount(props) {
  const hooks = [];
  let cursor, pending, changed, tree;
  const oldDocument = globalThis.document;
  globalThis.document = {
    activeElement: null,
    addEventListener() {},
    removeEventListener() {},
    querySelector() {
      return null;
    },
  };
  const hookDispatcher = {
    useState(initial) {
      const index = cursor++;
      if (!hooks[index])
        hooks[index] = {
          value: typeof initial === "function" ? initial() : initial,
        };
      return [
        hooks[index].value,
        (update) => {
          const next =
            typeof update === "function" ? update(hooks[index].value) : update;
          if (!Object.is(next, hooks[index].value)) {
            hooks[index].value = next;
            changed = true;
          }
        },
      ];
    },
    useRef(value) {
      const index = cursor++;
      return (hooks[index] ||= { current: value });
    },
    useMemo(fn, deps) {
      const index = cursor++;
      if (!hooks[index] || !equalDeps(hooks[index].deps, deps))
        hooks[index] = { value: fn(), deps };
      return hooks[index].value;
    },
    useEffect(fn, deps) {
      const index = cursor++;
      const previous = hooks[index];
      if (!previous || !equalDeps(previous.deps, deps)) {
        hooks[index] = { deps, cleanup: previous?.cleanup };
        pending.push(() => {
          hooks[index].cleanup?.();
          hooks[index].cleanup = fn();
        });
      }
    },
  };
  const render = (patch = {}) => {
    props = { ...props, ...patch };
    let passes = 0;
    do {
      assert.ok(
        ++passes < 20,
        "form effects must settle without a render loop",
      );
      cursor = 0;
      pending = [];
      changed = false;
      const previous = dispatcher.current;
      dispatcher.current = hookDispatcher;
      try {
        tree = Form(props);
      } finally {
        dispatcher.current = previous;
      }
      pending.forEach((effect) => effect());
    } while (changed);
    return tree;
  };
  const find = (predicate) => nodes(tree).find(predicate);
  const button = (label) =>
    find((node) => node.type === "button" && content(node) === label);
  render();
  return {
    render,
    find,
    button,
    text: () => content(tree),
    async settle() {
      await new Promise((resolve) => setImmediate(resolve));
      render();
    },
    close() {
      hooks.forEach((hook) => hook.cleanup?.());
      globalThis.document = oldDocument;
    },
  };
}
